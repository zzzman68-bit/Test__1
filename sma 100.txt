fastSma = sma(fastSource, fastLength)
medFastSma = sma(medFastSource, medFastLength)
mediumSma = sma(mediumSource, mediumLength)
slowSma = sma(slowSource, slowLength)
bullishCross = crossover(maExp, maSimp)
bullCross = crossover(fastSma, mediumSma)
bearishCross = crossunder(fastSma, mediumSma)
bullishCrossFast = crossover(fastSma, medFastSma)
bearishCrossFast = crossunder(fastSma, medFastSma)
fastBullCross = crossover(fastSma, slowSma)
fastBearCross = crossunder(fastSma, slowSma)
medBullCross = crossover(medFastSma, mediumSma)
expBullCross = crossover(fastSma, maExp)
simpBullCross = crossover(fastSma, maSimp)

ema = plot(maExp, title="100 EMA", color=color.new(color.green, 0), linewidth=1, style=plot.style_line)
sma = plot(maSimp, title="100 SMA", color=color.new(color.blue, 0), linewidth=1, style=plot.style_line)
